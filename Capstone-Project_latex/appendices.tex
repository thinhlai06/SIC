\begin{lstlisting}[language=Python, caption={Layer-wise learning-rate decay: parameter groups for a ViT (Equation~\eqref{eq:llrd}).}, label={lst:llrd}]
def param_groups(model, cfg, base_lr):
    """AdamW groups; layer-wise LR decay for ViT, no weight decay on 1-d params."""
    is_vit = hasattr(model, "blocks")
    n_layers = len(model.blocks) + 1 if is_vit else 0
    groups = {}
    for n, p in model.named_parameters():
        if not p.requires_grad: continue
        if is_vit:
            if n.startswith(("cls_token", "pos_embed", "patch_embed")): lid = 0
            elif n.startswith("blocks."): lid = int(n.split(".")[1]) + 1
            else: lid = n_layers
            scale = cfg["ld"] ** (n_layers - lid)
        else:
            scale = 1.0
        wd = 0.0 if (p.ndim == 1 or n.endswith(".bias")
                     or n in ("cls_token", "pos_embed")) else cfg["wd"]
        key = (scale, wd)
        groups.setdefault(key, {"params": [], "lr_scale": scale,
                                "weight_decay": wd})["params"].append(p)
    return list(groups.values())
\end{lstlisting}

\begin{lstlisting}[language=Python, caption={Square-root balanced sampler (Equation~\eqref{eq:sampler}) and warm-up plus cosine learning-rate schedule (Equation~\eqref{eq:cosine}).}, label={lst:sampler_lr}]
cnt = np.bincount(ytr, minlength=5).astype(float)
w = cnt[ytr] ** -0.5                      # sampling weight of every image
sampler = WeightedRandomSampler(torch.tensor(w / w.sum()),
                                num_samples=len(tr_items), replacement=True)

def lr_at(s):                             # s = optimisation step
    if s < warm:
        return base_lr * (s + 1) / max(1, warm)
    pr = (s - warm) / max(1, total - warm)
    return cfg["min_lr"] + (base_lr - cfg["min_lr"]) * 0.5 * (1 + math.cos(math.pi * pr))
\end{lstlisting}

\begin{lstlisting}[language=Python, caption={Validation metrics, with quadratic weighted kappa (Equation~\eqref{eq:qwk}) as the model-selection criterion.}, label={lst:metrics}]
def metrics(y, pred, logits=None):
    from sklearn.metrics import cohen_kappa_score, f1_score, recall_score
    r = recall_score(y, pred, labels=list(range(5)), average=None, zero_division=0)
    return dict(qwk=float(cohen_kappa_score(y, pred, weights="quadratic")),
                acc=float((y == pred).mean()),
                macro_f1=float(f1_score(y, pred, labels=list(range(5)),
                                        average="macro", zero_division=0)),
                recall=[float(x) for x in r])
\end{lstlisting}
